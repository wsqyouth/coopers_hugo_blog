{{- define "main" }} {{- $page := . -}} {{- /* 页面身份决定页头怎么写： · 首页 → 不出页头（直接是文章流），保持首屏干净 · term 页 → 出页头，标题用中文显示名（分类走 categoryNames 映射） · 其它列表 → 出页头，标题用页面自己的 Title */ -}} {{- $isTerm := eq .Kind "term" -}} {{- $title := .Title -}} {{- $eyebrow := "" -}} {{- $desc := "" -}} {{- if .IsHome -}} {{- /* 首页不显示 H1（站名已经在顶栏，重复一遍是噪音） */ -}} {{- $title = "" -}} {{- end -}} {{- if and $isTerm (eq .Type "categories") -}} {{- $title = index site.Params.categoryNames .Data.Term | default .Title -}} {{- end -}} {{- $pages := .RegularPages -}} {{- if .IsHome -}} {{- /* 用 Section 过滤而不是 Type —— Type 会被文章自己的 categories 覆盖（tech/thinking…），用它匹配 "posts" 会漏掉文章。 Section 稳定等于目录名 posts。 */ -}} {{- $pages = where site.RegularPages "Section" "in" site.Params.mainSections -}} {{- end -}} {{- if $pages -}} {{- $pages = $pages.ByDate.Reverse -}} {{- end -}} {{- $paginator := .Paginate $pages -}} {{- /* 正文区：文章流 注意：printf 产出的是普通字符串，直接输出会被 HTML 转义（<article）。 所以每一步累积都要过 safeHTML 保持它是 template.HTML。 */ -}} {{- $body := "" -}} {{- range $paginator.Pages -}} {{- $summary := partial "list-summary.html" . -}} {{- $entry := printf `

%s

%s

%s 阅读全文 →
` .Permalink (.Title | htmlEscape) ($summary | htmlEscape) (partial "article-meta.html" .) .Permalink (.Title | htmlEscape) -}} {{- $body = (printf "%s%s" $body $entry) | safeHTML -}} {{- end -}} {{- if gt $paginator.TotalPages 1 -}} {{- /* 分页链接不能用 cond —— Hugo 的 cond 两个分支都会求值， 没有上一页时访问 .Prev.URL 会命中 nil 而报错。 这里是两个独立的 if，各自只在自己成立时才求值。 */ -}} {{- $prev := `上一页` -}} {{- if $paginator.HasPrev -}} {{- $prev = printf `上一页` $paginator.Prev.URL -}} {{- end -}} {{- $next := `下一页` -}} {{- if $paginator.HasNext -}} {{- $next = printf `下一页` $paginator.Next.URL -}} {{- end -}} {{- $pg := printf `
%s 第 %d / %d 页 %s
` $prev $paginator.PageNumber $paginator.TotalPages $next -}} {{- $body = (printf "%s%s" $body $pg) | safeHTML -}} {{- end -}} {{ partial "list-shell.html" (dict "page" $page "title" $title "description" $desc "eyebrow" $eyebrow "body" $body ) }} {{- end }}
